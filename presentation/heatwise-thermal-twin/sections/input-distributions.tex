\subsection{Input distributions}

\begin{frame}[more-space]{Input Data\enspace|\enspace Input distributions}
  \dataheading{Input distributions with UQLab from ETH Z\"urich}
  {\footnotesize
  We used \textbf{UQLab.py 0.800}, the Python version of \textbf{UQLab from ETH Z\"urich},
  to create our input distributions.\par
  UQLab is a software framework for \textbf{uncertainty quantification}: it helps define
  input probabilities and study their effects on model outputs.\par
  Here, we assume \textbf{uniform} inputs with bounds from the \textbf{19 accepted period means}.
  \par\smallskip
  \renewcommand{\arraystretch}{1.2}
  \begin{tabular*}{\textwidth}{@{\extracolsep{\fill}}lrrl@{}}
    \toprule
    \textbf{Input} & \textbf{Lower bound} & \textbf{Upper bound} & \textbf{Unit}\\
    \midrule
    Total CPU power & 384.735 & 1,786.226 & W\\
    Inlet air temperature & 25.000 & 45.056 & ${}^\circ$C\\
    Raw fan signal & 0.121636 & 0.730571 & unverified\\
    \bottomrule
  \end{tabular*}
  }
  \medskip
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{beamercolorbox}[sep=2mm,wd=\linewidth]{data panel}
        \dataheading{A. Independent reference}
        \footnotesize
        Assume the three inputs are independent.\par
        Uniforms give a convenient \textbf{Legendre PCE basis}.\par
        Later UQ / Sobol results describe this chosen reference.
      \end{beamercolorbox}
    \column{0.49\textwidth}
      \begin{beamercolorbox}[sep=2mm,wd=\linewidth]{data panel}
        \dataheading{B. Optional dependent scenario}
        \footnotesize
        Keep the same uniform ranges, but specify dependence separately.\par
        A \textbf{Gaussian copula} describes an assumed scenario; it does not replace A.
      \end{beamercolorbox}
  \end{columns}
  \datasource{UQLab, ETH Z\"urich: \href{https://www.uqlab.com/}{www.uqlab.com}.
    Sources: results/inputs/reference\_model.json and assumptions.json.
    Displayed bounds rounded; saved parameters keep full precision. These are study assumptions, not physical uncertainty or safety limits.}
\end{frame}

\begin{frame}[more-space]{Input Data\enspace|\enspace Input distributions}
  \dataheading{Dependence is observed, but an operating probability law is not established}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{beamercolorbox}[sep=2mm,wd=\linewidth]{data panel}
        \dataheading{What the 19 observations tell us}
        \footnotesize
        Inlet / fan: \textbf{Pearson 0.826; Spearman 0.774}.\par
        \smallskip
        Five controlled cases with clustered inlet settings; these are not random samples of future operation.\par
        \smallskip
        \textbf{No dependent distribution is automatically fitted.}
      \end{beamercolorbox}
      \medskip
      {\footnotesize
      A Rosenblatt transformation is another option if conditional joint laws
      are known; it is not used here. Neither transformation creates missing coverage.
      }

    \column{0.49\textwidth}
      \begin{beamercolorbox}[sep=2mm,wd=\linewidth]{data panel}
        \dataheading{What optional model B does}
        \footnotesize
        A scenario file supplies the assumed correlation.\par
        \smallskip
        \textbf{Nataf transformation:} map inputs to normal scores, then remove the assumed correlation.
        The inverse maps back to the original units.\par
        \smallskip
        Matrix validity and forward / inverse round trips are tested. This checks the calculation, not the scenario's physical truth.
      \end{beamercolorbox}
      \medskip
      {\footnotesize
      Standard independent-input Sobol formulas apply to \textbf{A}; they cannot be reused unchanged for \textbf{B}'s physical inputs.
      }
  \end{columns}
  \datasource{Sources: results/inputs/report.md, assumptions.json and tables/input\_dependencies.csv; tests/test\_inputs.py.
    B assumes a Gaussian copula; exact uniform endpoints map to infinite normal scores and are rejected rather than clipped. No predictive model fitted yet.}
\end{frame}

\begin{frame}[more-space]{Input Data\enspace|\enspace Input distributions}
  \dataheading{Decision: follow model A with independent uniform inputs}
  \begin{beamercolorbox}[sep=2mm,wd=\textwidth]{data panel}
    \footnotesize
    For the first PCE, we choose \textbf{model A}: CPU power, inlet temperature and
    raw fan signal are \textbf{assumed mutually independent}, each with a uniform
    distribution over its observed range.
  \end{beamercolorbox}
  \medskip
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \dataheading{Why we chose this path}
      \footnotesize
      A clear, reproducible reference for studying the three inputs.\par
      \medskip
      Uniform distributions give a \textbf{Legendre PCE basis} and a standard
      independent-input Sobol analysis.\par
      \medskip
      The 19 controlled operating observations do not establish a reliable
      probability law for future operation.

    \column{0.49\textwidth}
      \dataheading{How we interpret this choice}
      \footnotesize
      The observed inlet / fan dependence \textbf{still exists in the experiment}.\par
      \medskip
      Independence is our \textbf{reference assumption}; it is not a finding
      about the measured data.\par
      \medskip
      Model B remains available for optional scenario studies, outside the
      chosen baseline.
  \end{columns}
  \medskip
  \begin{beamercolorbox}[sep=1.5mm,wd=\textwidth]{data warning}
    \footnotesize
    \textbf{Interpretation}\quad Later global UQ / Sobol results refer to model A's assumed domain.
    Predictions at untested input combinations still need validation.
  \end{beamercolorbox}
  \datasource{Project decision. Reference: results/inputs/reference\_model.json and assumptions.json.
    Input distributions created with UQLab.py, ETH Z\"urich: \href{https://www.uqlab.com/}{www.uqlab.com}.}
\end{frame}
